int partition(int f, int l) {
  int pivot = v[l];
  int i = f;

  for (int j = f; j < l; j++) 
    if (v[j] < pivot) 
      swap(i++,j);

  swap(i, l );
  return (i);
}
 
  

int qselect(int f, int l, int k) {
    int p = partition( f, l);

   
    if (p - 1 == k - 1){
        return v[p];
    }
  
    else if (k - 1 < p - 1){
        return qselect( f, p - 1, k);
    }
 
    else{
        return qselect(p + 1, l, k);
    }

    
   

}
